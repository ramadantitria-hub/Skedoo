🐘 Skema Supabase PostgreSQL (DDL Asli PDF)

-- 1. Tabel Profil Pengguna
create table profiles (
  id uuid references auth.users on delete cascade primary key,
  full_name text not null,
  nickname text,
  avatar_url text,
  gcal_access_token text,
  gcal_refresh_token text,
  push_subscription jsonb, -- Untuk Web Push Notification
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Tabel Duo Space (Ruang Berdua)
create table duo_spaces (
  id uuid default gen_random_uuid() primary key,
  name text default 'Our Space',
  pairing_code varchar(8) unique not null,
  user1_id uuid references profiles(id) on delete set null,
  user2_id uuid references profiles(id) on delete set null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Kategori Jadwal (Kuliah, Organisasi, Project, Belajar, dll)
create table categories (
  id uuid default gen_random_uuid() primary key,
  space_id uuid references duo_spaces(id) on delete cascade,
  name text not null,
  color_code text not null, -- e.g. #889C86, #7A919E
  icon_name text default 'calendar'
);

-- 4. Tabel Jadwal (Schedules)
create table schedules (
  id uuid default gen_random_uuid() primary key,
  space_id uuid references duo_spaces(id) on delete cascade not null,
  user_id uuid references profiles(id) on delete cascade not null,
  category_id uuid references categories(id) on delete set null,
  title text not null,
  description text,
  location text,
  start_time timestamp with time zone not null,
  end_time timestamp with time zone not null,
  is_recurring boolean default false,
  recurrence_pattern jsonb, -- e.g. {"type": "weekly", "days": [1, 3]}
  is_both boolean default false, -- Apakah ini agenda bersama?
  gcal_event_id text, -- ID sinkronisasi Google Calendar
  reminder_minutes integer default 15,
  sound_tone text default 'lofi_gentle', -- Tipe melodi notifikasi
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Mengaktifkan Supabase Realtime untuk sinkronisasi otomatis antar 2 orang
alter publication supabase_realtime add table schedules;
